from __future__ import annotations
import json, sqlite3, threading
from pathlib import Path
from typing import Any

class StateStore:
    def __init__(self, db_path: str = "veyra.db"):
        self.db_path = Path(db_path)
        self._lock = threading.Lock()
        self._init()

    def _connect(self):
        return sqlite3.connect(self.db_path, timeout=3)

    def _init(self):
        with self._connect() as con:
            con.execute("CREATE TABLE IF NOT EXISTS kv (key TEXT PRIMARY KEY, value TEXT NOT NULL)")
            con.execute("""CREATE TABLE IF NOT EXISTS events (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                ts DATETIME DEFAULT CURRENT_TIMESTAMP,
                kind TEXT NOT NULL,
                payload TEXT NOT NULL
            )""")
            con.commit()

    def set(self, key: str, value: Any):
        data = json.dumps(value)
        with self._lock, self._connect() as con:
            con.execute("INSERT INTO kv(key,value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value", (key, data))
            con.commit()

    def get(self, key: str, default=None):
        with self._connect() as con:
            row = con.execute("SELECT value FROM kv WHERE key=?", (key,)).fetchone()
        return default if row is None else json.loads(row[0])

    def log_event(self, kind: str, payload: dict):
        with self._lock, self._connect() as con:
            con.execute("INSERT INTO events(kind,payload) VALUES(?,?)", (kind, json.dumps(payload)))
            con.commit()

def emit(msg,raw=False):
    line=f"{datetime.now().isoformat(timespec='seconds')} {msg}"
    LOG.parent.mkdir(parents=True,exist_ok=True)
    with LOG.open('a',encoding='utf-8') as f:f.write(line+'\n')
    if raw:
        try: TTY2.write_text('',encoding='utf-8') if False else None
        except Exception: pass
        try:
            with TTY2.open('a',encoding='utf-8',errors='ignore') as f:f.write(line+'\n')
        except Exception: pass
    try:
        s=socket.socket(socket.AF_INET,socket.SOCK_DGRAM); s.settimeout(.05); s.sendto(line.encode()[:1400],REMOTE); s.close()
    except Exception: pass

def render():
    v=tracker.view(job); p=v['progress']; eta=p['eta_seconds']; pct=p['percent']; stage=v.get('stage') or 'Queued'
    if eta is None: eta_text='calculating ETA'; finish='unknown'
    else:
        eta_text=f"{int(eta//60)}m {int(eta%60)}s remaining"
        finish=(datetime.now()+timedelta(seconds=eta)).strftime('%H:%M')
    width=42; n=max(0,min(width,int(width*pct/100))); bar='█'*n+'░'*(width-n)
    screen=("\033[2J\033[H"+"VEYRA  v0.9.1 INSTALLATION\n"+"="*62+"\n\n"+f"Stage: {stage}\n\n[{bar}] {pct:5.1f}%\n\n{eta_text}   Predicted finish: {finish}\n\nCtrl+Alt+F2: live technical console\n")
    try:
        with TTY1.open('w',encoding='utf-8',errors='ignore') as f:f.write(screen)
    except Exception: pass

def run_stage(name,cmd,cwd=None):
    tracker.stage(job,name); emit(f'STAGE_START {name}',True)
    q=queue.Queue()
    p=subprocess.Popen(cmd,cwd=cwd,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True,bufsize=1)
    def reader():
        for line in p.stdout:
            q.put(line.rstrip())
    threading.Thread(target=reader,daemon=True).start()
    while p.poll() is None:
        while True:
            try: emit(q.get_nowait(),True)
            except queue.Empty: break
        render(); time.sleep(.5)
    while True:
        try: emit(q.get_nowait(),True)
        except queue.Empty: break
    if p.returncode:
        tracker.fail(job,f'{name} rc={p.returncode}'); emit(f'STAGE_FAIL {name} rc={p.returncode}',True); render(); raise SystemExit(p.returncode)
    emit(f'STAGE_PASS {name}',True); render()

def py_stage(name,fn):
    tracker.stage(job,name); emit(f'STAGE_START {name}',True); render()
    try: fn()
    except Exception as e:
        tracker.fail(job,str(e)); emit(f'STAGE_FAIL {name}: {e}',True); render(); raise
    emit(f'STAGE_PASS {name}',True); render()
