from __future__ import annotations
import json, sqlite3, threading
from pathlib import Path
from typing import Any

class StateStore:
    def __init__(self, db_path: str = "veyra.db"):
        self.db_path = Path(db_path)
        self._lock = threading.Lock()
        self._init()

    def _connect(self):
        return sqlite3.connect(self.db_path, timeout=3)

    def _init(self):
        with self._connect() as con:
            con.execute("CREATE TABLE IF NOT EXISTS kv (key TEXT PRIMARY KEY, value TEXT NOT NULL)")
            con.execute("""CREATE TABLE IF NOT EXISTS events (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                ts DATETIME DEFAULT CURRENT_TIMESTAMP,
                kind TEXT NOT NULL,
                payload TEXT NOT NULL
            )""")
            con.commit()

    def set(self, key: str, value: Any):
        data = json.dumps(value)
        with self._lock, self._connect() as con:
            con.execute("INSERT INTO kv(key,value) VALUES(?,?) ON CONFLICT(key) DO UPDATE SET value=excluded.value", (key, data))
            con.commit()

    def get(self, key: str, default=None):
        with self._connect() as con:
            row = con.execute("SELECT value FROM kv WHERE key=?", (key,)).fetchone()
        return default if row is None else json.loads(row[0])

    def log_event(self, kind: str, payload: dict):
        with self._lock, self._connect() as con:
            con.execute("INSERT INTO events(kind,payload) VALUES(?,?)", (kind, json.dumps(payload)))
            con.commit()

from pydantic import BaseModel, Field
from typing import Literal, Optional

class OverlayRequest(BaseModel):
    kind: Literal["text", "banner", "pip", "alert"] = "banner"
    title: str = ""
    message: str = ""
    priority: int = Field(default=1, ge=0, le=5)
    duration: int = Field(default=10, ge=1, le=300)
    video_url: Optional[str] = None
    image_url: Optional[str] = None
    position: Literal["top-left", "top-right", "bottom-left", "bottom-right", "center"] = "top-right"

class LaunchRequest(BaseModel):
    provider: str
    url: str
    title: str = ""
    season: Optional[int] = None
    episode: Optional[int] = None
    live: bool = False
    profile: str = "default"

class ComplianceAckRequest(BaseModel):
    profile: str = "default"
    acknowledged: bool = True

class LocalServerRequest(BaseModel):
    kind: str = "generic"
    name: str
    address: str

class ProvisionRequest(BaseModel):
    complete: Optional[bool] = None
    profile: Optional[str] = None
    device_name: Optional[str] = None
    ha_configured: Optional[bool] = None
    audio_sink: Optional[str] = None
    ui_scale: Optional[int] = Field(default=None, ge=80, le=160)

class AudioSinkRequest(BaseModel):
    sink: str

class ProviderSessionRequest(BaseModel):
    provider: str
    profile: str = "default"
    signed_in: bool = True

async def view_alive() -> bool:
    try:
        async with httpx.AsyncClient(timeout=0.35) as client:
            r = await client.get(f"http://{SETTINGS.view_bind}:{SETTINGS.view_port}/health")
            return r.status_code == 200
    except Exception:
        return False

async def send_view(message: dict) -> bool:
    # View failure must never block or crash Core. Commands are best-effort and journaled by callers.
    try:
        async with httpx.AsyncClient(timeout=0.45) as client:
            r = await client.post(SETTINGS.view_command_url, json=message, headers={"X-Veyra-Token": SETTINGS.view_token})
            return r.status_code == 200
    except Exception:
        store.log_event("view_delivery_failed", {"type": message.get("type")})
        return False


async def handle_ha_event(data: dict):
    try:
        req=OverlayRequest.model_validate(data)
    except Exception as exc:
        store.log_event("ha_overlay_rejected", {"error":str(exc)[:200]})
        return
    payload=apply_policy(req.model_dump(), store.get("context", {}))
    store.log_event("ha_overlay", payload)
    if not payload.get("suppressed"):
        await send_view({"type":"overlay","payload":payload})

    if ha_task:
        ha_task.cancel()
        try: await ha_task

@app.post("/api/launch")
async def launch(req: LaunchRequest):
    key=req.provider.lower().strip()
    if needs_tv_licence_notice(key, req.live) and not store.get(acknowledgement_key(req.profile), False):
        return {"ok":False,"requires_acknowledgement":True,"reason":"uk_tv_licence_notice"}
    adapter=PROVIDERS.get(key)
    if adapter is None:
        raise HTTPException(400, "Unknown provider")
    if not req.url.startswith("https://") or not adapter.validate_url(req.url):
        raise HTTPException(400, "URL is not permitted for this provider")
    payload=req.model_dump()
    payload["browser_profile_key"] = profiles.key(key, req.profile)
    payload["browser_profile_path"] = str(profiles.path_for(key, req.profile))
    store.set("last_launch", payload)
    store.log_event("launch", payload)
    if SETTINGS.provider_mode == "local":
        request=state_dir / "browser-control.json"
        tmp=request.with_suffix(".tmp")
        tmp.write_text(json.dumps({
            "provider": key,
            "url": req.url,
            "profile": payload["browser_profile_path"],
            "requested_at": __import__("time").time(),
        }))
        os.replace(tmp,request)
        return {"ok":True,"mode":"local"}
    await send_view({"type":"launch","payload":payload})
    return {"ok":True,"mode":"view"}

@app.post('/api/admin-console/set-pin')
async def admin_console_set_pin(payload: dict):
    pin=str(payload.get('pin','')).strip()
    old=str(payload.get('current_pin','')).strip()
    if not (pin.isdigit() and 4 <= len(pin) <= 12):
        raise HTTPException(400,'PIN must be 4-12 digits')
    current=store.get('admin_console_pin_hash','') or ''
    if current and not verify_pin(old,current):
        store.log_event('admin_console_pin_change_denied',{})
        raise HTTPException(403,'invalid current PIN')
    store.set('admin_console_pin_hash',hash_pin(pin))
    store.log_event('admin_console_pin_changed',{})
    return {'ok':True}

@app.post('/api/admin-console/open')
async def admin_console_open(payload: dict):
    stored=store.get('admin_console_pin_hash','') or ''
    pin=str(payload.get('pin',''))
    if not stored:
        raise HTTPException(409,'PIN not configured')
    if not verify_pin(pin,stored):
        store.log_event('admin_console_auth_failed',{})
        raise HTTPException(403,'invalid PIN')
    request=Path('/run/veyra-control/enter')
    try:
        request.parent.mkdir(parents=True,exist_ok=True)
        request.touch()
    except OSError:
        raise HTTPException(503,'local console broker unavailable')
    store.log_event('admin_console_open_requested',{})
    return {'ok':True,'handoff':'tty1','idle_timeout_s':900}

        except Exception: pass

def run_stage(name,cmd,cwd=None):
    tracker.stage(job,name); emit(f'STAGE_START {name}',True)
    q=queue.Queue()
    p=subprocess.Popen(cmd,cwd=cwd,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True,bufsize=1)
    def reader():
        for line in p.stdout:
            q.put(line.rstrip())
    threading.Thread(target=reader,daemon=True).start()
    while p.poll() is None:
        while True:
            try: emit(q.get_nowait(),True)
            except queue.Empty: break
        render(); time.sleep(.5)
    while True:
        try: emit(q.get_nowait(),True)
        except queue.Empty: break
    if p.returncode:
        tracker.fail(job,f'{name} rc={p.returncode}'); emit(f'STAGE_FAIL {name} rc={p.returncode}',True); render(); raise SystemExit(p.returncode)
    emit(f'STAGE_PASS {name}',True); render()
